# DarwinTV — Raspberry Pi 5 config with a separate kernel payload.
#
# The firmware loads loader8.img at 0x200000 and payload.bin at 0x08000000.
# Updating the kernel is then a single file copy; no reflash.

arm_64bit=1
kernel=loader8.img
kernel_address=0x200000

ramfsfile=payload.bin
ramfsaddr=0x08000000

uart_2ndstage=1

framebuffer_width=1920
framebuffer_height=1080
framebuffer_depth=32
